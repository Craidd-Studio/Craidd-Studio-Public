cmake_minimum_required(VERSION 3.16)
project(craidd_unity_probe LANGUAGES CXX)
set(CRAIDD_GOLD_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/anchored.cpp" CACHE FILEPATH "Explicit Gold anchored source or its prepared copy")
add_library(craidd_unity_probe SHARED probe.cpp "${CRAIDD_GOLD_SOURCE}")
target_include_directories(craidd_unity_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src-tauri/native/profiling")
target_compile_features(craidd_unity_probe PRIVATE cxx_std_17)
target_compile_options(craidd_unity_probe PRIVATE -g -O0 -fno-omit-frame-pointer -Wall -Wextra)
target_link_libraries(craidd_unity_probe PRIVATE pthread)
set_target_properties(craidd_unity_probe PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../Assets/Plugins/x86_64")
add_executable(native_smoke smoke.cpp)
target_link_libraries(native_smoke PRIVATE craidd_unity_probe)
enable_testing()
add_test(NAME native_interop COMMAND native_smoke)
